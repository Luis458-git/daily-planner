import { DAYS, dateKey, ordered } from '../Services/routineService'
import RoutineList from '../Components/RoutineList'
export default function Weekly({ activities, dates, selectedDay, setSelectedDay, actions }) {
  const selected = dates[selectedDay]
  return <><div className="page-heading"><div><p className="eyebrow">ORGANIZA TUS BLOQUES DE TIEMPO</p><h1>Rutina semanal</h1><p>{dates[0].toLocaleDateString('es-GT', { day: 'numeric', month: 'long' })} — {dates[6].toLocaleDateString('es-GT', { day: 'numeric', month: 'long', year: 'numeric' })}</p></div></div><section className="panel"><div className="week-grid">{DAYS.map((day, index) => <div className="week-column" key={day}><button className={`week-day ${selectedDay === index ? 'selected' : ''}`} onClick={() => setSelectedDay(index)} aria-pressed={selectedDay === index}>{day}<strong>{dates[index].getDate()}</strong></button>{ordered(activities.filter(a => a.days.includes(index))).map(a => <button className={`week-block category-${a.category}`} key={a.id} onClick={() => actions.onEdit(a)}><small>{a.start} – {a.end}</small><strong>{a.title}</strong><span>{a.completedDates.includes(dateKey(dates[index])) ? '✓ Completada' : a.category}</span></button>)}</div>)}</div></section><section className="panel"><h2>Actividades del {DAYS[selectedDay].toLowerCase()}</h2><RoutineList activities={activities.filter(a => a.days.includes(selectedDay))} date={dateKey(selected)} {...actions} /></section></>
}
