export default function TimerPanel({ timer, activities, small = false }) {
  const display = `${String(Math.floor(timer.remaining / 60)).padStart(2, '0')}:${String(timer.remaining % 60).padStart(2, '0')}`
  return <section className={`panel timer-panel ${small ? 'timer-small' : ''}`}><div className="section-heading"><h2>◷ {timer.kind === 'focus' ? 'Temporizador de enfoque' : 'Tiempo de descanso'}</h2><span className="badge">{timer.running ? 'En curso' : 'A tu ritmo'}</span></div>{!small && <div className="segmented">{[25, 45, 60].map(min => <button key={min} disabled={timer.locked} className={timer.kind === 'focus' && timer.total === min * 60 ? 'selected' : ''} onClick={() => timer.choose(min)}>{min} min</button>)}</div>}<label className="timer-activity">Actividad actual<select value={timer.activityId} disabled={timer.locked} onChange={e => timer.selectActivity(e.target.value)}><option value="">Enfoque libre</option>{activities.map(a => <option key={a.id} value={a.id}>{a.title}</option>)}</select></label><div className="timer-ring"><svg viewBox="0 0 200 200" aria-hidden="true"><circle cx="100" cy="100" r="90" /><circle cx="100" cy="100" r="90" pathLength="100" strokeDasharray="100" strokeDashoffset={100 - timer.remaining / timer.total * 100} /></svg><div><span className="eyebrow">{timer.kind === 'focus' ? 'ENFOQUE PROFUNDO' : 'RECARGA TU ENERGÍA'}</span><strong role="timer" aria-label="Tiempo restante">{display}</strong><small>{timer.running ? 'Un paso a la vez' : timer.locked ? 'Sesión en pausa' : 'Tu espacio. Tu ritmo.'}</small></div></div><div className="actions timer-actions"><button className="primary" onClick={timer.toggle}>{timer.running ? 'Ⅱ Pausar' : timer.locked ? '▷ Continuar' : '▷ Iniciar'}</button><button onClick={timer.addTime}>+5 min</button><button onClick={timer.finish}>Finalizar</button></div>{timer.message && <p className="success" role="status">{timer.message}</p>}</section>
}
